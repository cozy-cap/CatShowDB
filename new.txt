-- Create Table Cat
CREATE TABLE cat (
    cat_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    breed VARCHAR(100),
    owner_id INT,
    pedigree_number VARCHAR(50),
    is_pedigree_valid BOOLEAN
);

-- Create Table Vet
CREATE TABLE vet (
    vet_id SERIAL PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    degree VARCHAR(100),
    higher_education_institution VARCHAR(200),
    phone VARCHAR(30),
    email VARCHAR(100),
    vet_clinic VARCHAR(150),
    certificate_number VARCHAR(50),
    certificate_date DATE,
    certificate_expiration_date DATE
);

-- Create Table Vet_Check
CREATE TABLE vet_check (
    vet_check_id SERIAL PRIMARY KEY,
    cat_id INT REFERENCES cat(cat_id) ON DELETE CASCADE,
    check_date DATE NOT NULL,
    vaccination_all BOOLEAN,
    health_notes TEXT,
    is_healthy BOOLEAN,
    vet_id INT REFERENCES vet(vet_id) ON DELETE SET NULL
);
